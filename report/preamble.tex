% \mt{tenDinhDanh}: như \texttt nhưng cho phép ngắt dòng trong tên dài:
% sau dấu gạch dưới, giữa chữ thường và chữ hoa (camelCase), trước dấu mở
% ngoặc, sau các dấu . , : - /
\newcommand{\mtund}{\textunderscore\allowbreak}
\ExplSyntaxOn
\tl_new:N \l__mt_tl
\NewDocumentCommand{\mt}{m}
  {
    \group_begin:
    \tl_set:Nn \l__mt_tl {#1}
    \regex_replace_all:nnN { ([a-z0-9])([A-Z]) } { \1\c{allowbreak}\2 } \l__mt_tl
    \regex_replace_all:nnN { (\() } { \c{allowbreak}\1 } \l__mt_tl
    \regex_replace_all:nnN { ([\.\,\:\-\/]) } { \1\c{allowbreak} } \l__mt_tl
    \cs_set_eq:NN \_ \mtund
    \texttt{\l__mt_tl}
    \group_end:
  }
\ExplSyntaxOff
