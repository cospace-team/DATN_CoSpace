\section{Kiến trúc tổng thể}
\label{sec:kientruc}

Hệ thống theo mô hình khách -- chủ ba tầng như đã trình bày ở Chương~\ref{ch:cosolythuyet}. Hình~\ref{fig:kientruc_tongthe} thể hiện các thành phần và kênh giao tiếp giữa chúng.

\hinhdo{Chuong6/arch-tongthe}{Kiến trúc tổng thể của CoSpace}{fig:kientruc_tongthe}

\begin{enumerate}
    \item \textbf{Giao diện:} ứng dụng React được Vercel phân phối dưới dạng tệp tĩnh. Trình duyệt gọi API máy chủ qua HTTPS với token trong tiêu đề \mt{Authorization}; riêng trợ lý ảo nhận phản hồi dạng luồng Server-Sent Events (SSE).
    \item \textbf{Máy chủ ứng dụng:} Spring Boot chạy trong container Docker trên Render, gồm tầng bảo mật, controller, service, repository, các tác vụ định kỳ và bộ nhớ đệm Caffeine.
    \item \textbf{Cơ sở dữ liệu:} PostgreSQL trên Supabase, kết nối qua JDBC với bể kết nối HikariCP tới pooler của Supabase (cổng 6543, bắt buộc SSL).
    \item \textbf{Dịch vụ bên ngoài:} Supabase Auth (đăng nhập Google, cung cấp JWKS), PayOS và MoMo (tạo giao dịch, gửi webhook), Google Gemini (trợ lý ảo, lý do gợi ý đối tác).
\end{enumerate}

\subsection{Tổ chức mã nguồn phía máy chủ}
\label{ssec:phantang_be}
Mã nguồn máy chủ nằm trong gói \mt{com.cospace.app}:
\begin{itemize}
    \item \mt{controller}: 30 lớp REST controller với 167 điểm cuối; lớp \mt{GlobalExceptionHandler} (gói \mt{exception}) chuyển ngoại lệ thành phản hồi JSON.
    \item \mt{service}: 41 lớp, gồm các dịch vụ nghiệp vụ, ba tác vụ định kỳ, lớp \mt{BookingStateMachine} và lớp \mt{GeminiClient} gọi Gemini API. 29 lớp dùng \mt{@Transactional}.
    \item \mt{repository}: 34 giao diện Spring Data JPA, gồm các truy vấn khóa dòng \mt{findByIdWithLock} và truy vấn tìm khoảng thời gian chồng lấn.
    \item \mt{entity}: 34 lớp thực thể và 5 kiểu liệt kê (trạng thái đơn, trạng thái thanh toán, đơn vị thời gian, nguồn đơn, trạng thái bảo trì).
    \item \mt{dto}: 48 lớp dữ liệu vào/ra của API.
    \item \mt{config}: \mt{SecurityConfig}, \mt{SupabaseJwtAuthenticationConverter}, \mt{CacheConfig}, \mt{PayosConfig}, \mt{PayosProductionGuard}.
    \item \mt{security}: \mt{BranchAccessGuard}; \mt{util}: \mt{JwtUtil}.
\end{itemize}

\subsection{Tổ chức mã nguồn phía giao diện}
\label{ssec:phantang_fe}
Hình~\ref{fig:kientruc_frontend} thể hiện cấu trúc ứng dụng React.

\hinhdongang{Chuong6/arch-frontend}{Kiến trúc phía giao diện của CoSpace}{fig:kientruc_frontend}

\begin{itemize}
    \item \textbf{Định tuyến theo vai trò} (\mt{App.tsx}, React Router 7): mỗi vai trò có một nhóm trang riêng (\mt{/customer}, \mt{/staff}, \mt{/branch-admin}, \mt{/admin}); người dùng vào sai nhóm trang được chuyển về trang chủ của vai trò mình. Các trang được tải lười bằng \mt{React.lazy}.
    \item \textbf{Ngữ cảnh dùng chung:} \mt{AuthContext} giữ phiên đăng nhập, vai trò và chi nhánh; \mt{ThemeProvider} quản lý chế độ sáng/tối.
    \item \textbf{Tầng gọi API} (\mt{src/lib}, \mt{src/api}): các hàm gọi REST theo từng nhóm chức năng, tự gắn token và làm mới token khi nhận lỗi 401. Địa chỉ máy chủ được cấu hình qua biến \mt{VITE\_API\_BASE\_URL}.
    \item \textbf{Thành phần giao diện:} sơ đồ mặt bằng (xem và chỉnh sửa), khung chat trợ lý ảo, thông báo, các thành phần riêng của nhân viên và quản lý chi nhánh.
\end{itemize}

\section{Thiết kế lớp}
\label{sec:thietke_lop}

Các lớp thực thể không khai báo quan hệ đối tượng của Hibernate. Mỗi quan hệ được thể hiện bằng một thuộc tính kiểu \mt{UUID} giữ khóa ngoại (ký hiệu \guillemotleft FK\guillemotright{} trên sơ đồ). Mỗi sơ đồ chỉ nêu các thuộc tính và phương thức chính; lớp viền nét đứt đã được mô tả ở sơ đồ khác.

\subsection{Các lớp thực thể}
\label{ssec:lop_mien}

\subsubsection{Không gian làm việc}
Cấu trúc vật lý gồm \mt{BranchEntity} $\rightarrow$ \mt{Floor} $\rightarrow$ \mt{WorkspaceEntity}. Mỗi không gian thuộc một \mt{WorkspaceType}; tiện ích (\mt{Amenity}) được gắn theo loại không gian qua lớp \mt{WorkspaceTypeAmenity}. Ảnh của không gian là \mt{WorkspaceImage}; lịch bảo trì là \mt{WorkspaceMaintenanceEntity} (Hình~\ref{fig:cd_khonggian}).

\hinhuml{cd-01-khonggian}{Sơ đồ lớp thực thể: không gian làm việc}{fig:cd_khonggian}

\subsubsection{Bảng giá, dịch vụ thêm và chính sách hủy}
\mt{PricePolicy}, \mt{ExtraServiceEntity} và \mt{CancellationPolicy} đều có trường \mt{branchId} có thể rỗng: rỗng là cấu hình chung, có giá trị là cấu hình riêng của chi nhánh và được ưu tiên áp dụng. \mt{BranchServiceLimit} lưu số lượng tối đa của một dịch vụ được dùng cùng lúc ở một chi nhánh, tham chiếu dịch vụ bằng mã (Hình~\ref{fig:cd_gia}).

\hinhuml{cd-02-gia-chinhsach}{Sơ đồ lớp thực thể: bảng giá, dịch vụ thêm và chính sách hủy}{fig:cd_gia}

\subsubsection{Đặt chỗ, check-in, hủy và điểm uy tín}
\mt{Booking} lưu các khoản tiền \mt{subtotalAmount}, \mt{discountAmount}, \mt{addonAmount}, \mt{totalAmount} và phần giảm theo hạng thành viên, theo mã khuyến mãi. \mt{BookingServiceItem} là một dòng trên hóa đơn của đơn (dịch vụ thêm, phí gia hạn, phí trả muộn), lưu đơn giá tại thời điểm gọi và trạng thái đã trả hay chưa. \mt{CheckinLog} lưu các lượt check-in, check-out; \mt{BookingCancellation} lưu thông tin hủy. \mt{BookingGroup} gom các đơn của một lần đặt nhiều chỗ; \mt{ReputationEvent} ghi từng lần cộng, trừ điểm uy tín (\mt{User.reputationScore}) gắn với một đơn (Hình~\ref{fig:cd_datcho}).

\hinhuml{cd-03-datcho}{Sơ đồ lớp thực thể: đặt chỗ, check-in, hủy và điểm uy tín}{fig:cd_datcho}

\subsubsection{Thanh toán, hoàn tiền, khuyến mãi và hạng thành viên}
Một đơn có thể có nhiều \mt{Payment} (ví dụ lần thanh toán đơn và lần thanh toán dịch vụ phát sinh); các dòng \mt{Payment} của một lần trả chung cho đơn nhóm có cùng \mt{groupOrderId}. \mt{Refund} là một yêu cầu hoàn tiền trong hàng chờ. \mt{Promotion} lưu cả mã giảm giá công khai và voucher cá nhân (\mt{ownerUserId}); voucher cấp khi hoàn tiền được ghi ở \mt{Refund.voucherPromotionId}. Hạng của khách được lưu bằng mã hạng trong \mt{User}, tham chiếu tới \mt{MembershipTier} (Hình~\ref{fig:cd_thanhtoan}).

\hinhuml{cd-04-thanhtoan}{Sơ đồ lớp thực thể: thanh toán, hoàn tiền, khuyến mãi và hạng thành viên}{fig:cd_thanhtoan}

\subsubsection{Hồ sơ năng lực và gợi ý đối tác}
Mỗi \mt{User} có một \mt{Profile}. Kỹ năng (\mt{ProfileSkill}, có mức độ 1--5) và sở thích (\mt{ProfileInterest}, có độ ưu tiên) đều tham chiếu tới danh mục thẻ \mt{Tag}. \mt{ProfileMatchScore} lưu điểm tương đồng đã tính; \mt{PartnerConnection} lưu lời mời kết nối giữa hai người dùng (Hình~\ref{fig:cd_goiy}).

\hinhuml{cd-05-goiydoitac}{Sơ đồ lớp thực thể: hồ sơ năng lực và gợi ý đối tác}{fig:cd_goiy}

\subsubsection{Cộng đồng, thông báo và nhật ký kiểm toán}
Gồm bài viết (\mt{Post}), thẻ của bài viết (\mt{PostTag}), thông báo (\mt{NotificationEntity}) và nhật ký kiểm toán (\mt{AuditLogEntity}) (Hình~\ref{fig:cd_congdong}).

\hinhuml{cd-06-congdong}{Sơ đồ lớp thực thể: bảng tin, thông báo và nhật ký kiểm toán}{fig:cd_congdong}

\subsection{Các lớp dịch vụ}
\label{ssec:lop_dichvu}

\subsubsection{Đặt chỗ và tính giá}
\mt{BookingService} kiểm tra hạn mức đơn chờ và điểm uy tín (\mt{ReputationService}), lấy khóa tư vấn, lấy đơn giá từ \mt{PricingService}, áp dụng giảm giá qua \mt{MembershipService} và \mt{PromotionService}, tính giá dịch vụ thêm qua \mt{BookingAddonService}, lớp này kiểm tra giới hạn dịch vụ qua \mt{ServiceLimitService}; mọi chuyển trạng thái đi qua \mt{BookingStateMachine} (Hình~\ref{fig:cd_svc_datcho}).

\hinhumlngang{cd-07-svc-datcho}{Sơ đồ lớp dịch vụ: đặt chỗ, tính giá và ưu đãi}{fig:cd_svc_datcho}

\subsubsection{Vòng đời đơn, check-in và bảo trì}
\mt{BookingExpiryScheduler} (30 giây) giao từng đơn quá hạn cho \mt{BookingExpiryService}; \mt{BookingLifecycleScheduler} (5 phút) giao cho \mt{BookingLifecycleService} việc tự check-out, trừ điểm uy tín khi khách không check-in và đóng đơn hết giờ; \mt{BookingReminderScheduler} (1 phút) gửi nhắc lịch qua \mt{BookingReminderService}. \mt{CheckinService} xử lý check-in, check-out và cộng điểm khi check-in đúng giờ; \mt{StaffMaintenanceService} tạo lịch bảo trì và xử lý các đơn bị ảnh hưởng (Hình~\ref{fig:cd_svc_vongdoi}).

\hinhuml{cd-08-svc-vongdoi}{Sơ đồ lớp dịch vụ: vòng đời đơn, nhắc lịch, điểm uy tín, check-in và bảo trì}{fig:cd_svc_vongdoi}

\subsubsection{Thanh toán, hủy và hoàn tiền}
\mt{PaymentService} điều phối việc tạo giao dịch và xử lý webhook, gọi \mt{PayosService} và \mt{MomoService} để làm việc với cổng thanh toán. \mt{CancellationService} tính tiền hoàn khi hủy đơn, hủy đơn nhóm, hủy hoặc kết thúc sớm thay khách; \mt{RefundService} quản lý hàng chờ hoàn tiền và cấp voucher qua \mt{PromotionService}; \mt{NotificationService} gửi thông báo cho khách (Hình~\ref{fig:cd_svc_thanhtoan}).

\hinhumlngang{cd-09-svc-thanhtoan}{Sơ đồ lớp dịch vụ: thanh toán, hủy và hoàn tiền}{fig:cd_svc_thanhtoan}

\subsubsection{Xác thực và phân quyền}
\mt{AuthService} xử lý đăng ký, đăng nhập, làm mới token và liên kết tài khoản Google; \mt{JwtUtil} ký và kiểm tra token HS384; \mt{SupabaseJwtAuthenticationConverter} nạp vai trò từ cơ sở dữ liệu; \mt{BranchAccessGuard} giới hạn truy cập theo chi nhánh (Hình~\ref{fig:cd_svc_xacthuc}).

\mt{ChatbotService} điều phối hội thoại và các lần gọi hàm qua \mt{GeminiClient} (mô hình \mt{gemini-3.5-flash}, dự phòng \mt{gemini-3.7-flash}); \mt{PartnerMatchingService} tính điểm gợi ý; \mt{AuditLogService} ghi nhật ký kiểm toán trong một giao dịch riêng (\mt{REQUIRES\_NEW}), nên nhật ký vẫn được ghi ngay cả khi thao tác chính bị hoàn tác (Hình~\ref{fig:cd_svc_trolyao}).

\section{Sơ đồ tuần tự}
\label{sec:sequence}

Các sơ đồ tuần tự dưới đây mô tả những luồng xử lý chính, bám theo mã nguồn phía máy chủ; các luồng đơn giản hơn chỉ được mô tả bằng lời.

\subsection{Xác thực và phân quyền}
\begin{enumerate}
    \item \textbf{Đăng nhập (Hình~\ref{fig:sd_dangnhap}):} với email và mật khẩu, \mt{AuthService} kiểm tra mật khẩu bằng BCrypt và cấp token truy cập (1 giờ), token làm mới (30 ngày). Với Google, giao diện đăng nhập qua Supabase rồi gọi \mt{/api/auth/sync}; hệ thống chỉ tự liên kết với tài khoản khách hàng có cùng email khi token đến từ Google, và không bao giờ tự liên kết với tài khoản nhân viên hay quản trị.
    \item \textbf{Xác thực mỗi yêu cầu (Hình~\ref{fig:sd_xacthuc_api}):} bộ giải mã chọn cách kiểm tra theo nguồn phát hành token, converter đọc vai trò và trạng thái mới nhất từ cơ sở dữ liệu. Tài khoản bị khóa nhận lỗi 401 ngay ở yêu cầu kế tiếp.
\end{enumerate}

\hinhuml{sd-01-dangnhap}{Sơ đồ tuần tự đăng nhập bằng email hoặc Google và làm mới token}{fig:sd_dangnhap}
\hinhuml{sd-02-xacthuc-api}{Sơ đồ tuần tự xác thực và phân quyền một yêu cầu API}{fig:sd_xacthuc_api}

\subsection{Tạo đơn đặt chỗ}
\begin{enumerate}
    \item \textbf{Kiểm tra và khóa (Hình~\ref{fig:sd_datcho_1}):} trong cùng một giao dịch, \mt{BookingService} lấy khóa \mt{rate\_limit:<userId>} và đếm đơn chờ thanh toán (BR-03); suy ra chi nhánh từ không gian (BR-08); kiểm tra thời điểm và giờ mở cửa (BR-04, BR-05); lấy khóa \mt{booking:<workspaceId>}; tìm đơn chồng lấn, trong đó các đơn chờ đã quá hạn được giải phóng ngay; cuối cùng kiểm tra lịch bảo trì.
    \item \textbf{Tính giá và lưu (Hình~\ref{fig:sd_datcho_2}):} lấy đơn giá (giá chi nhánh trước, giá chung sau), tính số đơn vị ở máy chủ (BR-06), trừ giảm giá hạng thành viên rồi đến mã khuyến mãi, cộng dịch vụ thêm, lưu đơn \mt{PENDING\_PAYMENT} với hạn thanh toán 15 phút. Nếu tổng tiền bằng 0 (được giảm toàn bộ), đơn được xác nhận ngay.
\end{enumerate}

\hinhuml{sd-03-datcho-kiemtra}{Sơ đồ tuần tự tạo đơn (1/2): kiểm tra hạn mức, khóa và kiểm tra chồng lấn}{fig:sd_datcho_1}
\hinhuml{sd-04-datcho-luu}{Sơ đồ tuần tự tạo đơn (2/2): tính giá và lưu đơn}{fig:sd_datcho_2}

\subsection{Thanh toán}
\begin{enumerate}
    \item \textbf{Tạo giao dịch (Hình~\ref{fig:sd_thanhtoan}):} với PayOS hoặc MoMo, nếu đơn đã có một giao dịch đang chờ thì dùng lại giao dịch đó thay vì tạo mới; chỉ đơn \mt{PENDING\_PAYMENT} chưa quá hạn mới được thanh toán. Với tiền mặt, nhân viên xác nhận và giao dịch được tạo ngay ở trạng thái \mt{PAID}.
    \item \textbf{Xử lý webhook PayOS (Hình~\ref{fig:sd_webhook}):} kiểm tra chữ ký; bỏ qua nếu giao dịch đã \mt{PAID}; nếu thành công thì khóa dòng đơn rồi xác nhận đơn, hoặc tạo yêu cầu hoàn tiền khi tiền về muộn hay khách trả trùng (BR-15).
\end{enumerate}

\subsection{Hủy đơn và hoàn tiền}
\begin{enumerate}
    \item \textbf{Hủy đơn (Hình~\ref{fig:sd_huy}):} khóa dòng đơn; chỉ hủy được đơn chưa đến giờ bắt đầu (BR-11); tìm bậc chính sách phù hợp (BR-12); số tiền hoàn bằng phần trăm hoàn nhân tiền thuê cộng dịch vụ đã trả, không vượt số đã thu; phí phạt bằng tổng tiền trừ tiền hoàn (BR-13).
    \item \textbf{Xử lý hoàn tiền:} quản lý chi nhánh trả tiền cho khách rồi đánh dấu yêu cầu là đã xử lý, hoặc từ chối kèm lý do (BR-14).
\end{enumerate}

\hinhuml{sd-07-huy}{Sơ đồ tuần tự hủy đơn và tính tiền hoàn}{fig:sd_huy}

\subsection{Check-in, check-out và tác vụ định kỳ}
\begin{enumerate}
    \item \textbf{Check-in và check-out (Hình~\ref{fig:sd_checkin}):} kiểm tra chi nhánh, trạng thái, lượt check-in đang mở và khung giờ (BR-10); khi check-out, chặn nếu còn nợ dịch vụ (BR-18). Gói nhiều ngày quay về \mt{CONFIRMED} sau mỗi lần check-out cho đến khi hết hạn.
    \item \textbf{Giải phóng đơn quá hạn thanh toán (Hình~\ref{fig:sd_hethan}):} mỗi 30 giây; mỗi đơn được đọc lại dưới khóa dòng, nên đơn vừa được webhook xác nhận sẽ không bị chuyển sang \mt{EXPIRED}.
    \item \textbf{Đóng đơn hết giờ:} mỗi 5 phút, theo cùng cách với tác vụ giải phóng đơn; tự check-out khách quá giờ và chuyển đơn chưa từng check-in sang \mt{NO\_SHOW} (BR-17).
\end{enumerate}

\subsection{Máy trạng thái của giao dịch thanh toán}
Giao dịch thanh toán trực tuyến (\mt{PaymentStatus}) bắt đầu ở \mt{INITIATED}, chuyển sang \mt{PENDING} khi đã có liên kết thanh toán và sang \mt{PAID} khi cổng xác nhận thành công; khi chưa thu được tiền, giao dịch chuyển sang \mt{FAILED}, \mt{EXPIRED} hoặc \mt{CANCELLED}. \section{Thiết kế cơ sở dữ liệu}
\label{sec:thietke_csdl}

Cơ sở dữ liệu được thiết kế theo hai bước: mô hình thực thể -- kết hợp mức quan niệm theo ký pháp Chen \cite{chen1976er}, sau đó ánh xạ sang mô hình quan hệ.

\subsection{Mô hình thực thể -- kết hợp (ký pháp Chen)}
\label{ssec:erd_quan_niem}

Trong ký pháp Chen, thực thể là hình chữ nhật, mối kết hợp là hình thoi, thuộc tính là hình elip; thuộc tính khóa được gạch chân. Thực thể yếu và mối kết hợp xác định của nó được vẽ viền đôi. Trên mỗi cạnh nối thực thể với mối kết hợp ghi bản số $1$, $N$ hoặc $M$, ứng với các mối kết hợp $1:1$, $1:N$ và $M:N$. Mô hình được chia thành tám phần theo nhóm chức năng; thực thể có viền nét đứt đã được mô tả thuộc tính ở hình khác, và mỗi hình chỉ nêu các thuộc tính chính. Các cột chỉ dùng để ghi vết người thao tác (người tạo, người thu tiền, người hủy) và các cột lặp lại thông tin suy ra được từ đơn hoặc không gian (khách hàng, chi nhánh trên các bảng đặt chỗ, thanh toán, hoàn tiền) không được vẽ thành mối kết hợp.

\subsubsection{Định danh, hồ sơ năng lực và kết nối đối tác}
\emph{Người dùng} có nhiều \emph{Tài khoản liên kết} (nhà cung cấp đăng nhập) và một \emph{Hồ sơ năng lực}; hồ sơ là thực thể yếu, được xác định bởi người dùng sở hữu nó. Hồ sơ có quan hệ nhiều -- nhiều với \emph{Thẻ} qua hai mối kết hợp \emph{Có kỹ năng} (thuộc tính mức độ) và \emph{Quan tâm} (thuộc tính độ ưu tiên), và ghi nhận chi nhánh mà thành viên thường đến. Hai mối kết hợp đệ quy nhiều -- nhiều là \emph{Gợi ý} giữa hai hồ sơ (điểm tương đồng, lý do gợi ý) và \emph{Kết nối} giữa hai người dùng (trạng thái lời mời, lời nhắn) (Hình~\ref{fig:erd_chen_dinhdanh}).

\hinherd{erd-chen-01-dinhdanh.png}{ERD ký pháp Chen: định danh, hồ sơ năng lực và kết nối đối tác}{fig:erd_chen_dinhdanh}

\subsubsection{Không gian làm việc}
\emph{Chi nhánh} có nhiều \emph{Tầng}, mỗi tầng có nhiều \emph{Không gian}; nhân viên và quản lý chi nhánh làm việc tại một chi nhánh. Mỗi không gian thuộc một \emph{Loại không gian}; loại không gian có nhiều \emph{Tiện ích} (nhiều -- nhiều, kèm số lượng). Mỗi không gian có nhiều \emph{Ảnh không gian} và nhiều lịch \emph{Bảo trì} (Hình~\ref{fig:erd_chen_khonggian}).

\hinherd{erd-chen-02-khonggian.png}{ERD ký pháp Chen: không gian làm việc}{fig:erd_chen_khonggian}

\subsubsection{Bảng giá, dịch vụ thêm và hạng thành viên}
\emph{Bảng giá} tính cho một loại không gian và có thể áp dụng riêng cho một chi nhánh; bảng giá không gắn chi nhánh là giá chung. \emph{Dịch vụ thêm} do một chi nhánh cung cấp hoặc dùng chung. Với các dịch vụ là đồ vật (máy chiếu, màn hình di động), mỗi chi nhánh đặt số lượng tối đa được dùng cùng lúc qua mối kết hợp nhiều -- nhiều \emph{Giới hạn}. Mỗi đơn có nhiều \emph{Dòng phát sinh} (dịch vụ gọi thêm, phí gia hạn, phí trả muộn) lưu số lượng, đơn giá lúc ghi nhận và trạng thái thanh toán; dòng dịch vụ thuộc một dịch vụ thêm. Mỗi người dùng đạt một \emph{Hạng thành viên} (Hình~\ref{fig:erd_chen_gia}).

\hinherd{erd-chen-03-gia.png}{ERD ký pháp Chen: bảng giá, dịch vụ thêm và hạng thành viên}{fig:erd_chen_gia}

\subsubsection{Chính sách hủy và khuyến mãi}
\emph{Chính sách hủy} có loại quy tắc, ngưỡng nhỏ nhất, ngưỡng lớn nhất, phần trăm hoàn, độ ưu tiên và thời gian hiệu lực, có thể áp dụng riêng cho một chi nhánh. \emph{Khuyến mãi} có thể giới hạn theo chi nhánh và loại không gian; voucher cá nhân là khuyến mãi thuộc sở hữu của một người dùng (Hình~\ref{fig:erd_chen_chinhsach}).

\hinherd{erd-chen-04-chinhsach.png}{ERD ký pháp Chen: chính sách hủy và khuyến mãi}{fig:erd_chen_chinhsach}

\subsubsection{Đặt chỗ, check-in và hủy đơn}
\emph{Đặt chỗ} là thực thể trung tâm, nối \emph{Người dùng} với \emph{Không gian} trong một khoảng thời gian và có thể áp dụng một mã \emph{Khuyến mãi}. Mỗi đơn có nhiều \emph{Lượt check-in} do nhân viên thực hiện và có một \emph{Bản ghi hủy} khi bị hủy, lưu lý do, phần trăm và số tiền hoàn (Hình~\ref{fig:erd_chen_datcho}).

\hinherd{erd-chen-05-datcho.png}{ERD ký pháp Chen: đặt chỗ, check-in và hủy đơn}{fig:erd_chen_datcho}

\subsubsection{Thanh toán và hoàn tiền}
Mỗi đơn có nhiều \emph{Thanh toán}; mỗi thanh toán có nhiều \emph{Sự kiện thanh toán} và có thể dùng để trả các \emph{Dòng phát sinh}. \emph{Hoàn tiền} gắn với đơn và với giao dịch được hoàn, do một người dùng (quản lý chi nhánh hoặc quản trị viên) xử lý; khi hoàn bằng voucher, hệ thống cấp một \emph{Khuyến mãi} cá nhân (Hình~\ref{fig:erd_chen_thanhtoan}).

\hinherd{erd-chen-06-thanhtoan.png}{ERD ký pháp Chen: thanh toán và hoàn tiền}{fig:erd_chen_thanhtoan}

\subsubsection{Cộng đồng, thông báo và nhật ký kiểm toán}
\emph{Người dùng} viết nhiều \emph{Bài viết}; bài viết có thể đăng tại một chi nhánh và gắn nhiều \emph{Thẻ} (thuộc tính nguồn gắn thẻ). Người dùng nhận nhiều \emph{Thông báo}; mỗi \emph{Nhật ký kiểm toán} ghi người thực hiện, hành động, đối tượng, giá trị cũ và mới (Hình~\ref{fig:erd_chen_congdong}).

\hinherd{erd-chen-07-congdong.png}{ERD ký pháp Chen: cộng đồng, thông báo và nhật ký kiểm toán}{fig:erd_chen_congdong}

\subsubsection{Nhóm đặt chỗ và điểm uy tín}
Khách hàng có thể đặt nhiều chỗ trong một lần: \emph{Nhóm đặt chỗ} do một người dùng tạo, thuộc một chi nhánh, gồm nhiều đơn \emph{Đặt chỗ} cùng khoảng thời gian và được trả chung bằng một lần thanh toán. Mỗi người dùng có thuộc tính điểm uy tín; mỗi \emph{Sự kiện uy tín} phát sinh từ một đơn và ghi lý do, số điểm thay đổi và điểm sau khi thay đổi (Hình~\ref{fig:erd_chen_nhom}).

\hinherd{erd-chen-08-nhom-uytin.png}{ERD ký pháp Chen: nhóm đặt chỗ và điểm uy tín}{fig:erd_chen_nhom}

\subsection{Mô hình quan hệ}
\label{ssec:erd_vat_ly}

Mô hình quan niệm được ánh xạ sang mô hình quan hệ theo các nguyên tắc:
\begin{itemize}
    \item mỗi thực thể thành một bảng, khóa chính kiểu \mt{UUID} (\mt{gen\_random\_uuid()}); riêng thực thể yếu \emph{Hồ sơ năng lực} dùng mã người dùng làm khóa chính (\mt{profiles.user\_id});
    \item mối kết hợp $1:N$ thành khóa ngoại ở bảng phía nhiều; hai mối kết hợp $1:1$ thành khóa ngoại ở một phía: \emph{Hủy} thành \mt{booking\_cancellations.booking\_id} có ràng buộc duy nhất, \emph{Cấp voucher} thành \mt{refunds.voucher\_promotion\_id};
    \item mối kết hợp $M:N$ thành bảng trung gian: \mt{workspace\_type\_amenities}, \mt{profile\_skills}, \mt{profile\_interests}, \mt{profile\_match\_scores}, \mt{partner\_connections}, \mt{post\_tags}, \mt{branch\_service\_limits} (dịch vụ được tham chiếu bằng mã \mt{service\_code}, không qua khóa ngoại).
\end{itemize}
Hai mối kết hợp \emph{Làm việc tại} (người dùng -- chi nhánh) và \emph{Đạt hạng} (người dùng -- hạng thành viên) được lưu bằng cột \mt{users.branch\_id} và \mt{users.membership\_tier} nhưng chưa khai báo ràng buộc khóa ngoại.

Lược đồ được quản lý bằng Flyway với tám tệp migration: \mt{V1\_\_baseline.sql} tạo lược đồ gốc gồm 30 bảng; \mt{V2} bổ sung ràng buộc toàn vẹn; \mt{V3} thêm \mt{workspace\_images}, \mt{partner\_connections} và các cột cho phí phát sinh, voucher hoàn tiền; \mt{V4} thêm cột lưu sơ đồ tầng; \mt{V5} thêm điểm uy tín và bảng \mt{reputation\_events}; \mt{V6} thêm \mt{booking\_groups}; \mt{V7} thêm \mt{branch\_service\_limits}; \mt{V8} thêm lý do hoàn tiền \mt{STAFF\_ENDED}. Tổng cộng có 35 bảng, chia thành bảy nhóm (Hình~\ref{fig:erd} và Bảng~\ref{tab:bang_theo_module}). Mỗi đường nối trong hình cho biết giữa hai nhóm có ít nhất một khóa ngoại. Bảng \mt{users} và \mt{branches} được bảng của hầu hết các nhóm khác tham chiếu, nên hai nhóm chứa chúng được vẽ thành hai trục hai bên.

\subsection{Các ràng buộc toàn vẹn chính}
\begin{itemize}
    \item \textbf{Ràng buộc loại trừ trên \mt{bookings}:} \mt{EXCLUDE USING gist (workspace\_id WITH =, tstzrange(start\_at, end\_at) WITH \&\&)} chỉ áp dụng cho các đơn ở trạng thái \mt{PENDING\_PAYMENT}, \mt{CONFIRMED}, \mt{CHECKED\_IN} (BR-01). Bảng \mt{workspace\_maintenance} có ràng buộc tương tự cho lịch bảo trì ở trạng thái \mt{scheduled} và \mt{active}.
    \item \textbf{Công thức tiền của đơn (\mt{check\_booking\_amounts}):} các khoản tiền không âm và \mt{total\_amount = subtotal\_amount - discount\_amount + addon\_amount + tax\_amount + service\_fee\_amount}.
    \item \textbf{Hạn thanh toán (\mt{check\_payment\_deadline}):} đơn \mt{PENDING\_PAYMENT} phải có \mt{payment\_deadline\_at}.
    \item \textbf{Vai trò và chi nhánh (\mt{check\_branch\_by\_role}):} \mt{customer}, \mt{super\_admin} không có chi nhánh; \mt{staff}, \mt{branch\_admin} bắt buộc có chi nhánh.
    \item \textbf{Thời gian:} \mt{end\_at > start\_at} trên \mt{bookings}, \mt{workspace\_maintenance}, \mt{promotions}.
    \item \textbf{Chính sách hủy:} phần trăm hoàn trong $[0, 100]$ và $0 \le min \le max$.
    \item \textbf{Duy nhất:} mỗi đơn có tối đa một lượt check-in đang mở (\mt{uq\_checkin\_open}); mỗi giao dịch không được hoàn hai lần cho cùng một lý do (\mt{uq\_refunds\_payment\_reason}); mỗi cặp thành viên chỉ có một lời mời kết nối (\mt{uq\_partner\_connections\_pair}).
\end{itemize}
Các ràng buộc \mt{CHECK} của V2 được thêm ở chế độ \mt{NOT VALID}: PostgreSQL kiểm tra mọi dòng ghi mới nhưng không quét lại dữ liệu cũ, để migration chạy được trên cơ sở dữ liệu đang có dữ liệu. Sau khi dọn dữ liệu cũ, quản trị viên chạy \mt{VALIDATE CONSTRAINT} để bật kiểm tra đầy đủ.

\section{Thiết kế API}
\label{sec:thietke_api}

Máy chủ cung cấp 153 điểm cuối REST (tính theo cặp phương thức HTTP và đường dẫn) trong 28 controller. Quy ước chung:
\begin{itemize}
    \item Dữ liệu trao đổi dạng JSON; thời gian theo ISO-8601 có múi giờ.
    \item Quyền truy cập được kiểm tra ở hai mức: theo tiền tố đường dẫn trong \mt{SecurityConfig} và theo phương thức bằng \mt{@PreAuthorize}; các thao tác trên dữ liệu chi nhánh còn đi qua \mt{BranchAccessGuard}.
    \item Các điểm cuối tạo giao dịch PayOS và MoMo nhận tiêu đề tùy chọn \mt{Idempotency-Key}; khi đơn đã có một giao dịch đang chờ, hệ thống trả lại giao dịch đó thay vì tạo giao dịch mới.
    \item Điểm cuối \mt{POST /api/chatbot/message} trả về luồng SSE gồm các sự kiện tiến trình và sự kiện kết quả.
    \item Các điểm cuối công khai (không cần token) gồm: đăng ký, đăng nhập, làm mới token, kiểm tra sức khỏe \mt{/api/health}, danh sách chi nhánh và bảng giá tóm tắt, các điểm nhận webhook và trang trả về của PayOS, MoMo.
\end{itemize}

\section{Thiết kế bảo mật}
\label{sec:thietke_baomat}

Các biện pháp bảo mật được áp dụng ở nhiều lớp:
\begin{enumerate}
    \item \textbf{Mã hóa đường truyền:} giao diện và máy chủ được phục vụ qua HTTPS (do Vercel và Render cung cấp); kết nối tới cơ sở dữ liệu bắt buộc SSL.
    \item \textbf{Xác thực:} hai loại JWT như ở Mục~\ref{sec:cn_backend}; token làm mới không dùng được làm token truy cập; vai trò và trạng thái khóa luôn được đọc lại từ cơ sở dữ liệu.
    \item \textbf{Phân quyền theo vai trò:} quy tắc theo tiền tố đường dẫn trong \mt{SecurityConfig} và \mt{@PreAuthorize} ở từng phương thức.
    \item \textbf{Cô lập chi nhánh:} \mt{BranchAccessGuard} buộc nhân viên và quản lý chỉ thao tác trên chi nhánh của mình; quản trị viên hệ thống được truy cập mọi chi nhánh.
    \item \textbf{Xác thực webhook:} chữ ký HMAC-SHA256 của PayOS và MoMo được kiểm tra trước khi xử lý. Ở chế độ demo, PayOS từ chối mọi webhook; \mt{PayosProductionGuard} không cho ứng dụng chạy ở cấu hình production với khóa demo.
    \item \textbf{Chống tấn công chèn mã:} truy vấn cơ sở dữ liệu dùng tham số (JPA, \mt{JdbcTemplate}); giao diện làm sạch HTML bằng DOMPurify; Vercel trả về tiêu đề Content-Security-Policy có Trusted Types, \mt{X-Frame-Options: DENY} và \mt{X-Content-Type-Options: nosniff}.
    \item \textbf{CORS:} máy chủ chỉ chấp nhận các nguồn (origin) nằm trong danh sách cấu hình.
\end{enumerate}
Các điểm bảo mật còn hạn chế được trình bày ở Mục~\ref{sec:tk-han-che}.
