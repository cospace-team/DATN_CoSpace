\chapter{Tổng quan đề tài}
\label{ch:tongquan}

\section{Bối cảnh và động lực}
\label{sec:boicanh}

\subsection{Mô hình văn phòng chia sẻ}
Văn phòng chia sẻ (co-working space) cho thuê chỗ làm việc theo giờ, ngày, tuần hoặc tháng thay vì hợp đồng thuê văn phòng dài hạn. Người thuê không phải tự đầu tư nội thất, mạng Internet hay lễ tân, và có thể tăng giảm số chỗ theo nhu cầu. Mô hình này phù hợp với người làm việc tự do, nhóm khởi nghiệp và doanh nghiệp nhỏ, và đã xuất hiện ở nhiều quận trung tâm của TP. Hồ Chí Minh và Hà Nội.

\subsection{Thực trạng vận hành}
Qua khảo sát (Chương~\ref{ch:khaosat}), nhóm nhận thấy nhiều chuỗi co-working quy mô vừa và nhỏ vẫn vận hành bán thủ công và gặp các vấn đề sau:
\begin{itemize}
    \item \textbf{Không có sơ đồ trực quan:} tình trạng chỗ ngồi được ghi trên bảng tính, khách không biết vị trí cụ thể trước khi đến.
    \item \textbf{Đặt trùng lịch:} đặt chỗ đến từ nhiều kênh (Zalo, Fanpage, điện thoại, tại quầy) không đồng bộ nên hai khách có thể cùng đặt một chỗ.
    \item \textbf{Đối soát thanh toán thủ công:} nhân viên phải đối chiếu sao kê chuyển khoản; việc tính phí hủy và hoàn tiền chậm và dễ sai.
    \item \textbf{Chính sách hủy không thống nhất:} mỗi chi nhánh áp dụng một cách, không có công cụ cấu hình biểu phí theo thời gian.
    \item \textbf{Báo cáo rời rạc:} quản lý khó tổng hợp doanh thu và tỷ lệ sử dụng theo chi nhánh.
\end{itemize}

\subsection{Nhu cầu kết nối thành viên}
Một lý do khách chọn co-working thay vì văn phòng riêng là cơ hội gặp gỡ người làm việc cùng không gian. Tuy vậy, phần mềm quản lý hiện có chủ yếu phục vụ nghiệp vụ cho thuê. Đề tài bổ sung hồ sơ năng lực, gợi ý đối tác và bảng tin nội bộ để hỗ trợ nhu cầu này.

\section{Phát biểu bài toán}
\label{sec:baitoan}

Đề tài xây dựng hệ thống quản lý chuỗi văn phòng chia sẻ \textbf{CoSpace} phục vụ bốn nhóm người dùng: khách hàng (customer), nhân viên quầy (staff), quản lý chi nhánh (branch admin) và quản trị viên hệ thống (super admin).

Về nghiệp vụ, hệ thống cần hỗ trợ: xem sơ đồ mặt bằng và đặt chỗ, thanh toán trực tuyến qua VietQR và ví điện tử, check-in và check-out tại quầy, hủy đơn và hoàn tiền theo chính sách, kết nối thành viên và trợ lý ảo. Về kỹ thuật, hệ thống phải ngăn đặt trùng lịch khi có nhiều yêu cầu đồng thời, xử lý đúng khi cổng thanh toán gửi lặp thông báo, phân quyền theo vai trò và cô lập dữ liệu giữa các chi nhánh. \section{Mục tiêu đề tài}
\label{sec:muctieu}

\subsection{Mục tiêu nghiệp vụ}
\begin{enumerate}
    \item \textbf{Số hóa quy trình đặt chỗ:} tìm chỗ trên sơ đồ, giữ chỗ 15 phút chờ thanh toán, thanh toán trực tuyến và nhận mã đặt chỗ để check-in.
    \item \textbf{Quản lý nhiều chi nhánh:} phân cấp chi nhánh -- tầng -- không gian; bảng giá, dịch vụ và chính sách hủy riêng cho từng chi nhánh.
    \item \textbf{Tự động hóa hủy và hoàn tiền:} tính số tiền hoàn theo mốc thời gian, giải phóng chỗ ngay khi hủy.
    \item \textbf{Hỗ trợ vận hành quầy:} tạo đơn cho khách vãng lai, thu tiền mặt, check-in, check-out, ghi nhận dịch vụ phát sinh, gia hạn và tính phí trả chỗ muộn.
    \item \textbf{Kết nối cộng đồng:} hồ sơ năng lực, gợi ý đối tác, lời mời kết nối và bảng tin.
    \item \textbf{Báo cáo quản trị:} doanh thu, số đơn, tỷ lệ hủy và so sánh giữa các chi nhánh.
\end{enumerate}

\subsection{Mục tiêu kỹ thuật}
\begin{enumerate}
    \item \textbf{Kiến trúc phân tầng:} ứng dụng React/TypeScript, máy chủ Spring Boot 3, cơ sở dữ liệu PostgreSQL; phía máy chủ chia thành các tầng controller, service và repository.
    \item \textbf{Chống đặt trùng lịch:} kết hợp khóa ở tầng ứng dụng và ràng buộc ở tầng cơ sở dữ liệu.
    \item \textbf{Xử lý webhook thanh toán an toàn:} kiểm tra chữ ký và không ghi nhận lặp khi cổng thanh toán gửi lại.
    \item \textbf{Bảo mật và phân quyền:} hỗ trợ hai loại JWT (của Supabase khi đăng nhập Google và của hệ thống khi đăng nhập bằng email, ký HS384), phân quyền theo vai trò và cô lập dữ liệu chi nhánh.
    \item \textbf{Trợ lý ảo có kiểm soát:} dùng Google Gemini với cơ chế gọi hàm; thao tác ghi dữ liệu phải được người dùng xác nhận.
    \item \textbf{Triển khai trên nền tảng đám mây:} đóng gói máy chủ bằng Docker và chạy trên Render, giao diện trên Vercel, cơ sở dữ liệu trên Supabase.
\end{enumerate}

\section{Đối tượng và phạm vi}
\label{sec:phamvi}

\subsection{Đối tượng người dùng}
\begin{itemize}
    \item \textbf{Khách hàng:} tự đăng ký tài khoản; xem sơ đồ, đặt chỗ, thanh toán, hủy đơn, tham gia cộng đồng và dùng trợ lý ảo; có thể đặt chỗ ở mọi chi nhánh.
    \item \textbf{Nhân viên quầy:} thuộc một chi nhánh (\mt{branch\_id}); tạo đơn tại quầy, thu tiền mặt, check-in, check-out, ghi nhận dịch vụ phát sinh và khóa chỗ để bảo trì.
    \item \textbf{Quản lý chi nhánh:} thuộc một chi nhánh; cấu hình tầng, không gian, sơ đồ, bảng giá, chính sách hủy, dịch vụ; quản lý nhân viên, xử lý hoàn tiền và xem báo cáo của chi nhánh.
    \item \textbf{Quản trị viên hệ thống:} không gắn với chi nhánh; quản lý chi nhánh, danh mục dùng chung, người dùng, khuyến mãi, hạng thành viên, báo cáo toàn hệ thống và nhật ký kiểm toán.
\end{itemize}

\subsection{Phạm vi}
\begin{itemize}
    \item \textbf{Thị trường:} Việt Nam; giờ nghiệp vụ tính theo múi giờ \mt{Asia/Ho\_Chi\_Minh}; tiền tệ VNĐ, lưu dưới dạng số nguyên (kiểu \mt{long} trong Java).
    \item \textbf{Nền tảng:} ứng dụng web chạy trên trình duyệt máy tính và điện thoại.
    \item \textbf{Ngoài phạm vi:} ứng dụng di động riêng, tích hợp khóa cửa thông minh, tự động chi trả hoàn tiền qua cổng thanh toán. MoMo chỉ được tích hợp ở môi trường thử nghiệm (sandbox).
\end{itemize}

\section{Phương pháp và kế hoạch thực hiện}
\label{sec:phuongphap}

\subsection{Phương pháp}
\begin{itemize}
    \item \textbf{Nghiên cứu tài liệu:} đặc tả JWT (RFC 7519), OAuth 2.0 (RFC 6749), tài liệu Spring, PostgreSQL, PayOS, MoMo và Gemini API.
    \item \textbf{Khảo sát thực tế:} tìm hiểu quy trình đặt chỗ tại một số chuỗi co-working ở TP. Hồ Chí Minh và các phần mềm quản lý co-working phổ biến (Chương~\ref{ch:khaosat}).
    \item \textbf{Phát triển lặp:} chia công việc thành các vòng lặp ngắn, mỗi vòng có phần phát triển, kiểm thử và rà soát với giảng viên hướng dẫn.
\end{itemize}

\section{Ý nghĩa của đề tài}
\label{sec:ynghia}

\subsection{Ý nghĩa khoa học}
\begin{itemize}
    \item Áp dụng và kiểm chứng cách kết hợp khóa tư vấn mức giao dịch với ràng buộc loại trừ của PostgreSQL cho bài toán đặt tài nguyên theo khoảng thời gian.
    \item Áp dụng thiết kế xử lý lũy đẳng cho webhook thanh toán, kể cả các trường hợp tiền về sau khi đơn đã hết hạn hoặc khách trả trùng.
    \item Thử nghiệm việc đưa mô hình ngôn ngữ lớn vào hệ thống có giao dịch, trong đó mô hình chỉ được đề xuất thao tác ghi, còn việc thực hiện phải do người dùng xác nhận.
\end{itemize}

\subsection{Ý nghĩa thực tiễn}
\begin{itemize}
    \item Cung cấp một công cụ cho các chuỗi co-working vừa và nhỏ: giảm thao tác thủ công của lễ tân, tránh đặt trùng chỗ, thống nhất chính sách hủy giữa các chi nhánh.
    \item Giúp khách hàng chọn chỗ trên sơ đồ, thanh toán bằng mã VietQR và tìm người cộng tác trong cùng không gian làm việc.
\end{itemize}
