\chapter{Tổng quan đề tài}
\label{ch:tongquan}

\section{Bối cảnh và động lực thực hiện}
\label{sec:boicanh}

\subsection{Sự phát triển của mô hình văn phòng chia sẻ}
Trong bối cảnh kinh tế số và làn sóng khởi nghiệp tại Việt Nam, nhu cầu về môi trường làm việc linh hoạt ngày càng gia tăng. Theo các báo cáo của CBRE và Savills \cite{cbre_vietnam_2023, savills_vietnam_2023}, thị trường Co-working Space tại TP. Hồ Chí Minh và Hà Nội tăng trưởng bình quân trên 20\% mỗi năm trong giai đoạn hậu đại dịch. Xu hướng chuyển từ văn phòng truyền thống sang không gian làm việc linh hoạt đã trở thành lựa chọn phổ biến của Freelancers, Remote teams và SMEs.

Mô hình văn phòng chia sẻ giúp người thuê tiết kiệm chi phí đầu tư ban đầu (thiết kế mặt bằng, trang thiết bị, Internet, lễ tân) và linh hoạt chọn gói thuê theo giờ, ngày, tuần hoặc tháng thay vì hợp đồng dài hạn.

\subsection{Thực trạng vận hành của các chuỗi văn phòng chia sẻ}
Phần lớn các chuỗi Co-working quy mô vừa và nhỏ (2--10 chi nhánh) tại Việt Nam vẫn vận hành bán thủ công. Các khó khăn điển hình:
\begin{itemize}
    \item \textbf{Thiếu sơ đồ trực quan:} Thông tin chỗ ngồi được quản lý qua bảng tính; khách hàng không thể hình dung vị trí thực tế trước khi đến cơ sở.
    \item \textbf{Xung đột trùng lịch (Double Booking):} Nhận đặt chỗ qua nhiều kênh không đồng bộ (Zalo, Fanpage, điện thoại, tại quầy) dẫn đến hai khách cùng đặt một vị trí.
    \item \textbf{Đối soát thanh toán thủ công:} Chuyển khoản ngân hàng cần đối chiếu sao kê cuối ca; hủy đơn tính phạt và hoàn tiền chậm trễ, dễ sai sót.
    \item \textbf{Chính sách hủy không nhất quán:} Mỗi chi nhánh có chi phí mặt bằng khác nhau nhưng thiếu công cụ phân cấp biểu phí phạt hủy tự động.
    \item \textbf{Báo cáo phân mảnh:} Quản lý gặp khó khăn tổng hợp doanh thu và tỷ lệ lấp đầy theo thời gian thực.
\end{itemize}

\subsection{Nhu cầu kết nối cộng đồng}
Giá trị gia tăng cốt lõi của mô hình Co-working so với văn phòng truyền thống là \textbf{tính cộng đồng (Community Networking)}. Tuy nhiên, các phần mềm quản lý hiện có chỉ tập trung vào nghiệp vụ cho thuê, bỏ ngỏ bài toán kết nối thành viên. Nhu cầu xây dựng nền tảng hỗ trợ chia sẻ hồ sơ năng lực, tìm kiếm đối tác qua giải thuật so khớp và trao đổi trên bảng tin nội bộ là một động lực quan trọng của đề tài.

\section{Phát biểu bài toán}
\label{sec:baitoan}

Bài toán đặt ra: \textbf{Nghiên cứu, thiết kế và xây dựng hệ thống phần mềm quản lý văn phòng chia sẻ tập trung (CoSpace)}, phục vụ bốn nhóm tác nhân: Khách hàng (Customer), Nhân viên quầy (Staff), Quản lý chi nhánh (Branch Admin) và Quản trị viên hệ thống (Super Admin).

\section{Mục tiêu đề tài}
\label{sec:muctieu}

\subsection{Mục tiêu nghiệp vụ}
\begin{enumerate}
    \item \textbf{Số hóa vòng đời đặt chỗ:} Tìm kiếm, giữ chỗ 15 phút, thanh toán trực tuyến và nhận mã QR xác nhận tự động.
    \item \textbf{Quản lý chuỗi đa chi nhánh:} Phân tầng Tòa nhà -- Tầng -- Không gian; bảng giá và dịch vụ riêng từng chi nhánh.
    \item \textbf{Tự động hóa hủy và hoàn tiền:} Tính phí phạt và hoàn tiền theo mốc thời gian; giải phóng chỗ ngay cho khách khác.
    \item \textbf{Tối ưu vận hành quầy:} Tạo đơn walk-in, thu tiền mặt, check-in/check-out một chạm, ghi nhận dịch vụ phát sinh.
    \item \textbf{Kết nối cộng đồng:} Hồ sơ năng lực, gợi ý đối tác kinh doanh và diễn đàn nội bộ.
    \item \textbf{Báo cáo quản trị:} Dashboard doanh thu, tỷ lệ lấp đầy và so sánh hiệu quả giữa các chi nhánh.
\end{enumerate}

\subsection{Mục tiêu kỹ thuật}
\begin{enumerate}
    \item \textbf{Kiến trúc phân tầng:} Mô hình 3 tầng (React/TypeScript, Spring Boot 3, PostgreSQL) theo nguyên lý Clean Architecture.
    \item \textbf{Kiểm soát tương tranh:} Cơ chế khóa hai lớp độc lập ở tầng ứng dụng và CSDL, cam kết không xảy ra overbooking.
    \item \textbf{Webhook thanh toán bất biến:} Xác thực chữ ký mật mã, chống xử lý trùng lặp (Idempotent Consumer).
    \item \textbf{Bảo mật phân quyền:} Xác thực JWT kép (Supabase + nội bộ HS384), RBAC và cô lập dữ liệu chi nhánh.
    \item \textbf{Tích hợp AI có kiểm soát:} Trợ lý ảo Google Gemini với Function Calling, rào chắn xác nhận trước giao dịch ghi dữ liệu.
    \item \textbf{Triển khai đám mây:} Docker Multi-stage build, tự động phát hành lên Vercel, Render và Supabase.
\end{enumerate}

\section{Đối tượng và phạm vi đề tài}
\label{sec:phamvi}

\subsection{Đối tượng người dùng}
Hệ thống phục vụ bốn nhóm tác nhân:
\begin{itemize}
    \item \textbf{Khách hàng (Customer):} Đăng ký tự do, xem sơ đồ, đặt chỗ, thanh toán trực tuyến, check-in, hủy đơn, tham gia cộng đồng và tương tác trợ lý ảo. Có thể đặt chỗ tại bất kỳ chi nhánh nào.
    \item \textbf{Nhân viên quầy (Staff):} Gắn với một chi nhánh (\texttt{branch\_id}); xử lý đơn tại quầy, thu tiền mặt, check-in/out, ghi nhận dịch vụ phát sinh và quản lý lịch bảo trì.
    \item \textbf{Quản lý chi nhánh (Branch Admin):} Gắn với một chi nhánh; cấu hình không gian, sơ đồ SVG, bảng giá, biểu phí hủy, quản lý nhân viên, duyệt hoàn tiền và xem báo cáo nội bộ.
    \item \textbf{Quản trị viên hệ thống (Super Admin):} Không giới hạn chi nhánh; tạo chi nhánh, quản lý danh mục dùng chung, phân quyền toàn hệ thống, xem báo cáo tổng hợp và tra cứu Audit Log.
\end{itemize}

\subsection{Phạm vi}
\begin{itemize}
    \item \textbf{Địa lý và tiền tệ:} Tối ưu cho thị trường Việt Nam; múi giờ \texttt{Asia/Ho\_Chi\_Minh} (UTC+7); đơn vị VNĐ lưu trữ \texttt{numeric(12,2)} ánh xạ \texttt{BigDecimal} trong Java.
    \item \textbf{Nền tảng:} Ứng dụng Web Responsive, tương thích Chrome, Firefox, Edge, Safari trên máy tính bàn, máy tính bảng và điện thoại.
    \item \textbf{Giới hạn:} Tập trung vào phần mềm Web và backend; ứng dụng di động bản địa và phần cứng IoT (khóa cửa thông minh) là hướng mở rộng tương lai.
\end{itemize}

\section{Ý nghĩa đề tài}
\label{sec:ynghia}

\subsection{Ý nghĩa khoa học}
\begin{itemize}
    \item \textbf{Kiểm soát tương tranh phân tán:} Cung cấp mô hình thực nghiệm kết hợp \texttt{pg\_advisory\_xact\_lock} và \texttt{EXCLUDE USING gist} trên PostgreSQL để giải quyết xung đột tài nguyên hữu hạn mà không gây khóa chết (Deadlock).
    \item \textbf{Tích hợp thanh toán bất biến:} Đưa ra giải pháp xử lý Webhook bất đồng bộ từ bên thứ ba, đảm bảo toàn vẹn dữ liệu tài chính khi mạng có độ trễ cao hoặc gửi lặp sự kiện.
    \item \textbf{Kiểm soát LLM trong hệ thống giao dịch:} Áp dụng Prompt Engineering và Function Calling với cơ chế Confirmation Barrier, mở hướng tiếp cận an toàn khi tích hợp AI tạo sinh vào hệ thống vận hành.
\end{itemize}

\subsection{Ý nghĩa thực tiễn}
\begin{itemize}
    \item Cung cấp giải pháp chuyển đổi số cho các doanh nghiệp vận hành Co-working Space tại Việt Nam, giảm thiểu thời gian xử lý thủ công của nhân viên lễ tân và loại bỏ tổn thất do đặt trùng phòng.
    \item Nâng cao trải nghiệm khách hàng thông qua sơ đồ mặt bằng trực quan và thanh toán tức thời qua chuẩn VietQR Napas 24/7.
\end{itemize}

\subsection{Ý nghĩa xã hội}
\begin{itemize}
    \item Hỗ trợ phong trào khởi nghiệp sáng tạo và lực lượng lao động tự do (Freelancers) tiếp cận môi trường làm việc chuyên nghiệp với chi phí linh hoạt, tối ưu nguồn lực mặt bằng xã hội.
    \item Kiến tạo nền tảng kết nối cộng đồng tri thức số, giúp các nhóm dự án và cá nhân tìm kiếm cộng sự, mở rộng mạng lưới hợp tác kinh doanh.
\end{itemize}
