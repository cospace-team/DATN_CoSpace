\section{Cơ sở lý thuyết}
\label{sec:coso_lythuyet}

\subsection{Kiến trúc ứng dụng web nhiều tầng}
CoSpace theo mô hình khách -- chủ ba tầng, phía máy chủ được chia tiếp thành các lớp theo trách nhiệm \cite{martin2017clean}:
\begin{itemize}
    \item \textbf{Tầng trình bày:} ứng dụng một trang (Single Page Application -- SPA) viết bằng React và TypeScript, chạy trên trình duyệt; hiển thị dữ liệu, vẽ sơ đồ mặt bằng và thu nhận thao tác người dùng. Tầng này không chứa quy tắc nghiệp vụ.
    \item \textbf{Tầng ứng dụng:} máy chủ Spring Boot nhận yêu cầu HTTPS, xác thực, phân quyền, thực thi quy tắc nghiệp vụ và quản lý giao dịch.
    \item \textbf{Tầng dữ liệu:} PostgreSQL trên Supabase, lưu trữ dữ liệu và thực thi các ràng buộc toàn vẹn.
\end{itemize}

\hinhdongang{Chuong3/arch-phantang}{Kiến trúc phân tầng của CoSpace}{fig:kientruc_phantang}

Phía máy chủ gồm ba lớp:
\begin{enumerate}
    \item \textbf{Controller:} nhận yêu cầu REST, kiểm tra định dạng dữ liệu vào bằng Jakarta Validation, kiểm tra quyền và gọi tầng dịch vụ. Controller không chứa quy tắc nghiệp vụ.
    \item \textbf{Service:} chứa quy tắc nghiệp vụ, ranh giới giao dịch (\mt{@Transactional}), khóa tương tranh và chuyển trạng thái đơn qua lớp \mt{BookingStateMachine}.
    \item \textbf{Repository:} dùng Spring Data JPA để đọc và ghi dữ liệu, gồm cả các truy vấn khóa dòng (\mt{SELECT \ldots{} FOR UPDATE}).
\end{enumerate}

\subsection{Kiến trúc REST}
API của hệ thống theo phong cách REST \cite{fielding2000rest}:
\begin{itemize}
    \item \textbf{Không lưu trạng thái phiên:} mỗi yêu cầu mang token trong tiêu đề \mt{Authorization: Bearer <token>}, máy chủ không giữ phiên đăng nhập trong bộ nhớ.
    \item \textbf{Định danh tài nguyên bằng URI:} ví dụ \mt{/api/bookings}, \mt{/api/payments}, \mt{/api/refunds}. Một số thao tác nghiệp vụ không phải CRUD được biểu diễn bằng URI con dạng động từ, ví dụ \mt{POST /api/bookings/\{id\}/cancel-v2} hay \mt{POST /api/refunds/\{id\}/process}.
    \item \textbf{Phương thức HTTP:} \mt{GET} để đọc; \mt{POST} để tạo hoặc thực hiện thao tác nghiệp vụ; \mt{PUT}, \mt{PATCH} để cập nhật; \mt{DELETE} để xóa hoặc ngừng sử dụng. Theo RFC 9110, \mt{GET}, \mt{PUT} và \mt{DELETE} là các phương thức lũy đẳng \cite{rfc9110}.
    \item \textbf{Mã trạng thái:} \mt{200}/\mt{201} khi thành công; \mt{400} khi dữ liệu không hợp lệ hoặc vi phạm quy tắc (ví dụ chỗ đã có người đặt); \mt{401} khi thiếu hoặc sai token; \mt{403} khi không đủ quyền; \mt{409} khi thao tác không phù hợp với trạng thái hiện tại (ví dụ đã có ba đơn chờ thanh toán); \mt{500} cho lỗi không lường trước.
\end{itemize}

\subsection{Xác thực bằng JWT}
Hệ thống dựa trên OAuth 2.0 \cite{rfc6749} và JSON Web Token (JWT) \cite{rfc7519}. Một JWT gồm ba phần \mt{Header.Payload.Signature}: phần đầu ghi thuật toán ký, phần thân chứa các claim như \mt{sub} (mã người dùng), \mt{iat}, \mt{exp}, \mt{iss}, và phần chữ ký. Với thuật toán HS384, chữ ký được tính bằng
\begin{equation}
    Signature = \text{HMAC-SHA384}\big(\text{base64url}(Header) \,\|\, \text{``.''} \,\|\, \text{base64url}(Payload),\ secret\big)
    \label{eq:jwt}
\end{equation}

CoSpace chấp nhận hai loại token:
\begin{enumerate}
    \item \textbf{Token của Supabase Auth} khi người dùng đăng nhập bằng Google. Token được ký bằng khóa bất đối xứng trên đường cong P-256; máy chủ kiểm tra chữ ký bằng khóa công khai lấy từ điểm cuối JWKS của Supabase.
    \item \textbf{Token do CoSpace cấp} khi đăng nhập bằng email và mật khẩu. Mật khẩu được băm bằng BCrypt; token truy cập có hạn 1 giờ, token làm mới có hạn 30 ngày, cả hai được ký HS384 với khóa bí mật dài tối thiểu 48 byte.
\end{enumerate}

\subsection{Giao dịch và điều khiển tương tranh}
Cơ sở dữ liệu quan hệ bảo đảm các tính chất ACID cho giao dịch \cite{kleppmann2017ddia}. Tuy vậy, nếu ứng dụng ``đọc rồi mới ghi'' mà không khóa, hai giao dịch chạy song song có thể cùng thấy một chỗ còn trống và cùng tạo đơn, dẫn đến đặt trùng (Hình~\ref{fig:race_condition}).

\hinhdo{Chuong3/race-condition}{Hai khách cùng đặt một chỗ: không có bảo vệ, chỉ có ràng buộc EXCLUDE, và khóa tư vấn kết hợp EXCLUDE}{fig:race_condition}

CoSpace dùng hai cơ chế của PostgreSQL \cite{postgresql_documentation}:
\begin{enumerate}
    \item \textbf{Khóa tư vấn mức giao dịch (\mt{pg\_advisory\_xact\_lock}):} khóa đặt trên một số nguyên do ứng dụng quy định, không khóa bảng hay dòng. Khóa được tính bằng \mt{hashtext('booking:' || workspaceId)}. Hai giao dịch đặt cùng một chỗ nhận cùng khóa nên phải chạy lần lượt; khóa tự nhả khi giao dịch kết thúc. Vì \mt{hashtext} trả về số nguyên 32 bit, hai chỗ khác nhau có thể trùng khóa; khi đó giao dịch chỉ phải chờ thêm chứ không làm sai kết quả.
    \item \textbf{Ràng buộc loại trừ (exclusion constraint):} dùng phần mở rộng \mt{btree\_gist} để khai báo
\begin{verbatim}
EXCLUDE USING gist (workspace_id WITH =,
                    tstzrange(start_at, end_at) WITH &&)
WHERE (status IN ('PENDING_PAYMENT','CONFIRMED','CHECKED_IN'))
\end{verbatim}
    Ràng buộc này bảo đảm không có hai đơn đang hiệu lực cùng chỗ mà khoảng thời gian giao nhau. Nó là lớp chặn cuối cùng nếu một luồng xử lý nào đó bỏ qua khóa tư vấn.
\end{enumerate}

\subsection{Tính lũy đẳng khi xử lý webhook thanh toán}
Cổng thanh toán (PayOS, MoMo) báo kết quả giao dịch cho máy chủ bằng webhook, tức một yêu cầu HTTP POST \cite{payos_docs}. Nếu không nhận được phản hồi kịp thời, cổng thanh toán gửi lại, nên máy chủ có thể nhận cùng một thông báo nhiều lần. Một thao tác $f$ là \textbf{lũy đẳng} (idempotent) nếu thực hiện nhiều lần cho cùng kết quả như thực hiện một lần:
\begin{equation}
    f(f(x)) = f(x)
    \label{eq:idempotency}
\end{equation}

CoSpace xử lý webhook như sau: kiểm tra chữ ký HMAC-SHA256; tìm giao dịch theo mã đơn; nếu giao dịch đã ở trạng thái \mt{PAID} thì bỏ qua và trả \mt{200}; ngược lại cập nhật giao dịch rồi khóa dòng đơn đặt chỗ bằng \mt{SELECT \ldots{} FOR UPDATE} trước khi xác nhận đơn. Nếu tiền về khi đơn đã hết hạn hoặc đã bị hủy, hoặc khách đã thanh toán đơn đó bằng một giao dịch khác, hệ thống tạo yêu cầu hoàn tiền thay vì xác nhận đơn. Chỉ mục duy nhất trên cặp (\mt{payment\_id}, \mt{reason\_type}) của bảng \mt{refunds} ngăn tạo hai yêu cầu hoàn tiền cho cùng một giao dịch và lý do.

\subsection{Hệ số Jaccard và điểm gợi ý đối tác}
Hệ số Jaccard đo độ giống nhau của hai tập hợp \cite{jaccard1912}:
\begin{equation}
    J(A, B) = \frac{|A \cap B|}{|A \cup B|}
    \label{eq:jaccard_formula}
\end{equation}

CoSpace tính điểm gợi ý giữa người dùng $u$ và ứng viên $c$ từ ba tín hiệu:
\begin{itemize}
    \item $s_1 = J(K_u, K_c)$: độ giống nhau của tập kỹ năng;
    \item $s_2 = J(I_u, I_c)$: độ giống nhau của tập sở thích;
    \item $s_3 = \dfrac{|T_c \cap T_u|}{\min(|T_c|, |T_u|)}$: mức trùng giữa chủ đề các bài viết của ứng viên ($T_c$) và các chủ đề người dùng quan tâm ($T_u$, gồm kỹ năng, sở thích và chủ đề bài viết của chính người dùng).
\end{itemize}
Mỗi tín hiệu $k$ có trọng số $w_1 = 0{,}5$, $w_2 = 0{,}2$, $w_3 = 0{,}2$ và chỉ được tính khi cả hai phía đều có dữ liệu ($\delta_k = 1$). Điểm cuối cùng là
\begin{equation}
    S(u, c) = \min\left(1,\ \frac{\sum_k w_k\, \delta_k\, s_k}{\sum_k w_k\, \delta_k} + 0{,}15 \cdot b(u, c)\right)
    \label{eq:weighted_jaccard}
\end{equation}
trong đó $b(u, c) = 1$ nếu hai người có cùng chi nhánh làm việc chính, ngược lại bằng 0. Việc chia cho tổng trọng số của các tín hiệu có dữ liệu giúp một thành viên chưa khai báo kỹ năng vẫn nhận được điểm có ý nghĩa từ các tín hiệu còn lại.

\subsection{Mô hình ngôn ngữ lớn và gọi hàm}
Trợ lý ảo dùng Google Gemini qua cơ chế gọi hàm (function calling) \cite{gemini_function_calling}. Máy chủ gửi kèm danh sách hàm dưới dạng lược đồ JSON; thay vì chỉ trả lời bằng văn bản, mô hình có thể trả về yêu cầu gọi một hàm với tham số trích từ câu hỏi của người dùng. Máy chủ thực thi hàm, gửi kết quả lại cho mô hình và lặp cho đến khi mô hình đưa ra câu trả lời.

Để mô hình không tự ý thay đổi dữ liệu, các hàm được chia làm hai nhóm. Hàm chỉ đọc (tra cứu chi nhánh, chỗ trống, đơn đặt chỗ, chính sách) được thực thi ngay. Hàm ghi (đặt chỗ, hủy chỗ, gọi thêm dịch vụ) chỉ tạo một đề xuất; giao diện hiển thị thẻ xác nhận và thao tác chỉ được thực hiện khi người dùng bấm xác nhận.

\section{Công nghệ phía giao diện}
\label{sec:cn_frontend}

\subsection{React 18 và TypeScript 5}
Giao diện dùng React 18.3 \cite{react_docs} và TypeScript 5.6. React chia màn hình thành các thành phần (component) tái sử dụng được; TypeScript kiểm tra kiểu ở thời điểm biên dịch (dự án bật chế độ \mt{strict}). Định tuyến dùng React Router 7, các trang được tải lười bằng \mt{React.lazy} để giảm dung lượng tải ban đầu.

\subsection{Vite và Tailwind CSS}
Vite 5.4 \cite{vite_docs} dùng ES module gốc của trình duyệt khi phát triển và dùng Rollup khi đóng gói. Cấu hình \mt{manualChunks} tách các thư viện dùng chung (React, Supabase, Framer Motion, react-icons) thành các tệp riêng để trình duyệt lưu đệm độc lập với mã ứng dụng. Giao diện dùng Tailwind CSS 3.4 \cite{tailwind_docs} và hỗ trợ chế độ sáng/tối qua lớp \mt{dark} trên phần tử gốc.

\subsection{Hiển thị sơ đồ mặt bằng}
Bố cục mỗi tầng được lưu dưới dạng JSON gồm kích thước khung vẽ, kích thước ô lưới và danh sách phần tử (bàn, phòng, tường, cửa, thang máy\ldots). Mỗi phần tử có vị trí, kích thước, góc xoay và có thể gắn với một không gian (\mt{workspaceId}). Thành phần \mt{FloorPlanViewer} vẽ bố cục này thành SVG và tô màu các phần tử có gắn không gian theo tình trạng tại khung giờ đang chọn: xanh lá là còn trống, đỏ là đã có người đặt, xám là đang bảo trì. Người dùng có thể phóng to, thu nhỏ và kéo sơ đồ. Vì là đồ họa vector, sơ đồ hiển thị rõ ở mọi độ phân giải.

\subsection{Các thư viện khác}
\begin{itemize}
    \item \textbf{Recharts:} vẽ biểu đồ doanh thu và cơ cấu đơn trên các trang báo cáo.
    \item \textbf{react-icons:} bộ biểu tượng dùng trong toàn bộ giao diện.
    \item \textbf{Framer Motion:} hiệu ứng chuyển cảnh.
    \item \textbf{html5-qrcode:} quét mã QR đặt chỗ bằng camera trên trang check-in.
    \item \textbf{DOMPurify:} làm sạch nội dung HTML trước khi hiển thị, kết hợp chính sách Trusted Types trong tiêu đề Content-Security-Policy.
\end{itemize}

\section{Công nghệ phía máy chủ}
\label{sec:cn_backend}

\subsection{Spring Boot 3 và Spring Security}
Máy chủ dùng Spring Boot 3.1.12 \cite{spring_boot_docs}, mã nguồn biên dịch cho Java 17 và chạy trên JRE 21 trong container. Spring Security 6 được cấu hình như một máy chủ tài nguyên OAuth 2.0: mỗi yêu cầu được xác thực từ token, không dùng phiên đăng nhập. Bộ giải mã token (\mt{JwtDecoder}) đọc trường \mt{iss} để chọn cách kiểm tra: token của Supabase được kiểm tra bằng JWKS, token của CoSpace được kiểm tra bằng khóa HS384. Sau đó lớp \mt{SupabaseJwtAuthenticationConverter} đọc lại vai trò, chi nhánh và trạng thái tài khoản từ cơ sở dữ liệu, nên vai trò ghi trong token không thể dùng để leo thang quyền và tài khoản bị khóa mất quyền truy cập ngay ở yêu cầu tiếp theo.

\subsection{Spring Data JPA và bộ nhớ đệm Caffeine}
Tầng truy cập dữ liệu dùng Spring Data JPA với Hibernate 6. Các thực thể tham chiếu nhau bằng trường khóa ngoại kiểu UUID thay vì các quan hệ \mt{@ManyToOne}/\mt{@OneToMany}; cách làm này tránh truy vấn N+1 và lỗi nạp lười ngoài giao dịch, đổi lại các truy vấn cần dữ liệu liên quan phải được viết tường minh.

Các danh mục ít thay đổi (chi nhánh, loại không gian, bảng giá, dịch vụ thêm, chính sách hủy, tiện ích, hạng thành viên) và một số truy vấn quản trị được lưu đệm bằng Caffeine \cite{caffeine_github}. Mỗi bộ đệm có thời gian sống từ 1 đến 10 phút, tối đa 500 phần tử, và được xóa (\mt{@CacheEvict}) khi dữ liệu tương ứng bị sửa.

\subsection{Tác vụ định kỳ}
Hai tác vụ định kỳ dùng \mt{@Scheduled}:
\begin{itemize}
    \item \mt{BookingExpiryScheduler} chạy mỗi 30 giây, tìm các đơn \mt{PENDING\_PAYMENT} đã quá hạn thanh toán và chuyển sang \mt{EXPIRED} để giải phóng chỗ.
    \item \mt{BookingLifecycleScheduler} chạy mỗi 5 phút: tự check-out các đơn \mt{CHECKED\_IN} đã quá giờ kết thúc 120 phút, và đóng các đơn \mt{CONFIRMED} đã hết giờ thành \mt{COMPLETED} (nếu đã từng check-in) hoặc \mt{NO\_SHOW}.
\end{itemize}
Mỗi đơn được xử lý trong một giao dịch riêng và được đọc lại dưới khóa dòng, nên lỗi ở một đơn không làm dừng các đơn còn lại và không ghi đè kết quả của webhook đến cùng lúc.

\section{Cơ sở dữ liệu và dịch vụ bên ngoài}
\label{sec:cn_database}

\subsection{PostgreSQL trên Supabase}
Cơ sở dữ liệu là PostgreSQL \cite{postgresql_documentation} do Supabase quản lý \cite{supabase_docs}. Các tính năng được dùng:
\begin{itemize}
    \item Kiểu khoảng thời gian có múi giờ (\mt{tstzrange}) và toán tử giao nhau \mt{\&\&}.
    \item Phần mở rộng \mt{btree\_gist} để kết hợp phép so sánh bằng trên \mt{workspace\_id} với phép giao khoảng thời gian trong một ràng buộc loại trừ.
    \item Khóa tư vấn mức giao dịch \mt{pg\_advisory\_xact\_lock}.
\end{itemize}
Lược đồ được quản lý bằng Flyway: \mt{V1\_\_baseline.sql} là lược đồ gốc, \mt{V2\_\_constraints.sql} bổ sung các ràng buộc toàn vẹn, \mt{V3} thêm dòng phí phát sinh, voucher hoàn tiền, ảnh không gian và bảng kết nối đối tác. Supabase còn cung cấp dịch vụ đăng nhập Google và kho lưu ảnh không gian (Supabase Storage).

\subsection{Cổng thanh toán PayOS và ví MoMo}
PayOS \cite{payos_docs} tạo mã VietQR chứa sẵn số tài khoản, số tiền và nội dung chuyển khoản. Khi khách chuyển khoản thành công, PayOS gửi webhook về máy chủ; dữ liệu được ký bằng HMAC-SHA256 trên chuỗi \mt{key=value} của các trường, sắp theo thứ tự bảng chữ cái, với khóa checksum của cửa hàng. Khi chưa cấu hình khóa thật, dịch vụ PayOS chạy ở chế độ demo: mọi webhook bị từ chối, việc thanh toán được giả lập qua một điểm cuối riêng. Lớp \mt{PayosProductionGuard} không cho ứng dụng khởi động ở cấu hình production nếu PayOS vẫn ở chế độ demo.

MoMo \cite{momo_docs} được tích hợp ở môi trường sandbox: máy chủ tạo yêu cầu thanh toán có chữ ký HMAC-SHA256 và kiểm tra chữ ký của thông báo IPN do MoMo gửi về.

\subsection{Google Gemini API}
Máy chủ gọi Gemini qua REST (\mt{generateContent}) \cite{google_gemini_api}. Tên mô hình được cấu hình qua biến môi trường; cấu hình mặc định dùng \mt{gemini-3.5-flash} và chuyển sang mô hình dự phòng \mt{gemini-3.7-flash} khi mô hình chính quá tải hoặc từ chối yêu cầu. Gemini được dùng cho trợ lý ảo và để viết câu giải thích ngắn cho mỗi gợi ý đối tác.
